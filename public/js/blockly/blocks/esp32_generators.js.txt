 * Traduce la estructura gráfica del bloque en código equivalente de Arduino.
 */

// Usamos una referencia dinámica para soportar múltiples versiones de Blockly
const generator = Blockly.JavaScript || javascript.javascriptGenerator;

// 1. Traductor para Configurar Pin (pinMode)
generator['esp32_pin_mode'] = function(block) {
  const dropdown_pin = block.getFieldValue('PIN');
  const dropdown_mode = block.getFieldValue('MODE');
  return `pinMode(${dropdown_pin}, ${dropdown_mode});\n`;
};

// 2. Traductor para Escritura Digital (digitalWrite)
generator['esp32_digital_write'] = function(block) {
  const dropdown_pin = block.getFieldValue('PIN');
  const dropdown_state = block.getFieldValue('STATE');
  return `digitalWrite(${dropdown_pin}, ${dropdown_state});\n`;
};

// 3. Traductor para Lectura Analógica (analogRead)
generator['esp32_analog_read'] = function(block) {
  const dropdown_pin = block.getFieldValue('PIN');
  const code = `analogRead(${dropdown_pin})`;
  // Importante: Al retornar un valor, se especifica el orden de precedencia atómica
  return [code, generator.ORDER_ATOMIC || 0];
};
