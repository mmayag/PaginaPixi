// 1. Bloque para configurar el modo del Pin (pinMode)
Blockly.Blocks['esp32_pin_mode'] = {
  init: function() {
    this.appendDummyInput()
        .appendField("Configurar Pin GPIO#")
        .appendField(new Blockly.FieldDropdown([
            ["2","2"],["4","4"],["5","5"],["12","12"],["13","13"],
            ["14","14"],["15","15"],["16","16"],["17","17"],["18","18"],
            ["19","19"],["21","21"],["22","22"],["23","23"],["25","25"],
            ["26","26"],["27","27"],["32","32"],["33","33"]
        ]), "PIN")
        .appendField("como")
        .appendField(new Blockly.FieldDropdown([
            ["SALIDA (OUTPUT)", "OUTPUT"],
            ["ENTRADA (INPUT)", "INPUT"],
            ["ENTRADA PULLUP", "INPUT_PULLUP"]
        ]), "MODE");
    this.setPreviousStatement(true, null);
    this.setNextStatement(true, null);
    this.setColour("#4ec9b0"); // Tu color --accent-green de styles.css
    this.setTooltip("Configura un pin físico de la ESP32 como entrada o salida.");
    this.setHelpUrl("");
  }
};

// 2. Bloque para escribir digitalmente (digitalWrite)
Blockly.Blocks['esp32_digital_write'] = {
  init: function() {
    this.appendDummyInput()
        .appendField("Establecer Pin GPIO#")
        .appendField(new Blockly.FieldDropdown([
            ["2","2"],["4","4"],["5","5"],["12","12"],["13","13"],
            ["14","14"],["15","15"],["17","17"],["21","21"],["22","22"],
            ["23","23"],["25","25"],["26","26"],["27","27"]
        ]), "PIN")
        .appendField("en estado")
        .appendField(new Blockly.FieldDropdown([
            ["ALTO (HIGH / 1)", "HIGH"],
            ["BAJO (LOW / 0)", "LOW"]
        ]), "STATE");
    this.setPreviousStatement(true, null);
    this.setNextStatement(true, null);
    this.setColour("#4ec9b0");
    this.setTooltip("Envía un estado lógico de 3.3V (HIGH) o 0V (LOW) a un pin.");
    this.setHelpUrl("");
  }
};

// 3. Bloque para lectura analógica (analogRead)
Blockly.Blocks['esp32_analog_read'] = {
  init: function() {
    this.appendDummyInput()
        .appendField("Leer Pin Analógico ADC GPIO#")
        .appendField(new Blockly.FieldDropdown([
            ["32","32"],["33","33"],["34","34"],["35","35"],["36","36"],["39","39"]
        ]), "PIN");
    this.setOutput(true, "Number");
    this.setColour("#4ec9b0");
    this.setTooltip("Lee el valor de voltaje analógico (0-4095) en pines compatibles con ADC.");
    this.setHelpUrl("");
  }
};
